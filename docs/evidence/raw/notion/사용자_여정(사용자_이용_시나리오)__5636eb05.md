    ```
flowchart TD
    nodeEntry(["그래프에서 노드 선택"])
    pageEntry(["AI 챗봇 페이지 직접 진입"])

    nodeEntry --> askButton["이 노드에 질문하기 클릭"]
    askButton --> context["선택한 기업·이슈와 추천 질문 확인"]
    pageEntry --> chatHome["대화 화면과 추천 질문 확인"]

    context --> input["추천 질문 선택 또는 직접 질문 입력"]
    chatHome --> input
    input --> submit["질문 전송"]

    submit --> intent{"서비스가 질문 의도 판단"}

    intent -->|"관계·연결 경로 탐색"| graphAnswer["관계 그래프와 연결 설명 확인"]
    intent -->|"개념·요약·수치 설명"| textAnswer["일반 텍스트 답변 확인"]

    graphAnswer --> nextAction{"사용자의 다음 행동"}
    textAnswer --> nextAction

    nextAction -->|"추가 질문"| followUp(["대화 맥락을 유지하여 다음 질문 입력"])
    nextAction -->|"근거 확인"| evidence["뉴스·공시 출처 선택 및 원문 확인"]
    evidence --> returnChat(["기존 대화로 복귀"])
    nextAction -->|"질문 마치기"| finish(["이전 화면 복귀 또는 이용 종료"])

    %% 후속 질문은 실제로 질문 입력 단계로 이어지지만,
    %% 배치가 복잡해지지 않도록 복귀 화살표 없이 연결 지점으로 표시
    %% 의도 판단은 사용자 선택이 아닌 서비스 내부 처리
    %% 그래프와 텍스트 답변 모두 근거 출처 제공
    %% 질문이 모호하면 추가 확인 요청
    %% 근거가 부족하면 확인 가능한 내용과 한계를 안내

    classDef default fill:#EDE9FE,stroke:#8B7AC4,color:#25203D;
    classDef entry fill:#D1FAE5,stroke:#32936A,color:#164E38;
    classDef decision fill:#FFDEB0,stroke:#D99538,color:#663C10;
    classDef system fill:#DBEAFE,stroke:#6092CB,color:#1E4068;
    classDef finish fill:#FECACA,stroke:#D66A6A,color:#7F1D1D;

    class nodeEntry,pageEntry,followUp,returnChat entry;
    class nextAction decision;
    class intent system;
    class finish finish;
```

  → 시장 전체의 어느 요소를 보여줄까?? 
  1. 거래대금의 변화율이 가장 큰 종목의 이슈와 종목간의 관계 위주→ 이거롤안함
  1. 이슈의 우선순위를 바탕으로 그것에 영향을 받는 종목 위주→ 이거 위주로 표시
  이슈의 우선순위를 어떻게 판단??
  이슈가 하나 있을 때, 여기서 뻗어나가는 간선의 수가 많은 이슈 
  → 두 방식을 섞어 로직을 만든다.
  발원지 이슈부터 기업까지 연결되는 하나의 경로를 보여준다.
